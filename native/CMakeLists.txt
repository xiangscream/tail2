cmake_minimum_required(VERSION 3.20)
project(tail2_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LIBDEV_ROOT "$ENV{LIBDEV_ROOT}" CACHE PATH "Private SDK root containing include/dev/devs.hpp")
find_path(BOOST_HEADERS boost/json/src.hpp HINTS "$ENV{BOOST_ROOT}" PATH_SUFFIXES include REQUIRED)
if(NOT EXISTS "${LIBDEV_ROOT}/include/dev/devs.hpp")
  message(FATAL_ERROR "Set LIBDEV_ROOT to the extracted SDK directory; SDK must stay outside Git")
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use an x64 toolchain")
  endif()
  set(SDK_BIN "${LIBDEV_ROOT}/windows/win64-release")
  set(SDK_LIB "${SDK_BIN}/libdev.lib")
elseif(UNIX AND NOT APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  set(SDK_BIN "${LIBDEV_ROOT}/linux/x86_64-release")
  set(SDK_LIB "${SDK_BIN}/libdev.so")
else()
  message(FATAL_ERROR "M0 build supports Windows x64 and Linux x86_64 only")
endif()
find_package(Threads REQUIRED)
add_executable(tail2_probe main.cpp)
target_compile_definitions(tail2_probe PRIVATE BOOST_JSON_NO_LIB)
target_include_directories(tail2_probe PRIVATE "${LIBDEV_ROOT}/include" "${BOOST_HEADERS}")
target_link_libraries(tail2_probe PRIVATE "${SDK_LIB}" Threads::Threads)
if(MSVC)
  target_compile_options(tail2_probe PRIVATE /utf-8 /W4)
  set_property(TARGET tail2_probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  foreach(dll libdev.dll w32-pthreads.dll)
    add_custom_command(TARGET tail2_probe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${SDK_BIN}/${dll}" "$<TARGET_FILE_DIR:tail2_probe>")
  endforeach()
else()
  target_compile_options(tail2_probe PRIVATE -Wall -Wextra)
  set_target_properties(tail2_probe PROPERTIES BUILD_RPATH "${SDK_BIN}")
endif()
